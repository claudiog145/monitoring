// KPI ingestion - exceptions of one service of a company profile (Spec 1000 contract v4 §2,
// AppInsightsKpiSource).
//
// Nothing in this file names a company, a host, a step or a customer. CompanyKqlTemplater fills
// the request and exception source unions, host filter, entity identity and service/step
// classifier placeholders from the definition's ProfileKey, scoped to the service named by the
// domain token: the domain covers every host of that service. IKqlSourceResolver then turns each
// source alias into a resource id. (Placeholders are never written inside a comment: a multi-line
// fragment would end the comment.) AppInsightsKpiSource validates and formats every
// other token: from/to as datetime literals, bucket as a timespan, perBucket and maxEntities as
// integers, stage and domain as quoted strings (stage empty = every stage) and grain as the bare
// word domain or entity.
//
// An exception inherits the stage (and entity) of the classified request with the same
// operation in the same source - the way the Scout board attributes it. Browser ErrorEvent noise
// is dropped.
// Domain grain output (contract v1): only kind "bucket" rows. TotalCount = sum(itemCount) of
// exceptions, Value = exceptions as a percentage of the requests of the same bucket.
// Entity grain output: only kind "bucket" rows, one per entity x bucket, keeping the maxEntities
// entities with the most requests per bucket. EntityKey is the entity alone; the stage is the
// definition's Stage, and an empty Stage aggregates every stage of the service per entity
// (contract v4 §6 A1).
// TotalCount = exceptions, Value = exceptions as a percentage of requests, FailedCount = failed
// requests, P50/P95/Mean = null. No samples.
let fromTs = {{from}};
let toTs = {{to}};
let bucketSize = {{bucket}};
// Exceptions produce no sample rows; the token is kept so every KPI template shares one contract.
let samplesPerBucket = {{perBucket}};
let stageFilter = {{stage}};
let serviceKey = {{domain}};
let maxEntities = {{maxEntities}};
let historyStart = fromTs;
let tEnd = toTs;
let unknownEntity = "(entity unavailable)";
let reqs = materialize(
    union
{{SOURCE_UNION}}
    | where timestamp >= fromTs and timestamp < toTs
    | extend Host = tolower(tostring(parse_url(url).Host))
    | where {{HOST_FILTER}}
{{SERVICE_AND_STEP_CLASSIFIER}}
    | where ServiceKey == serviceKey and isnotempty(Step)
    | extend Stage = Step
    | where isempty(stageFilter) or Stage == stageFilter
    | extend BucketStart = bin_at(timestamp, bucketSize, datetime(1970-01-01))
    | extend Entity = tostring({{ENTITY_IDENTITY}})
    | extend Entity = case(isnotempty(Entity), Entity, isnotempty(Host), Host, unknownEntity)
    | project BucketStart, timestamp, itemCount, success, Ref = strcat(Source, ":", operation_Id),
              HasOperation = isnotempty(operation_Id), Entity, Stage);
let requestMap =
    reqs
    | where HasOperation
    | summarize arg_max(timestamp, Entity, Stage) by Ref
    | project Ref, RequestEntity = Entity, RequestStage = Stage;
let stageExceptions =
    union
{{EXCEPTION_SOURCE_UNION}}
    | where timestamp >= fromTs and timestamp < toTs
    | where type !startswith "ErrorEvent"
    | where isnotempty(operation_Id)
    | extend Ref = strcat(Source, ":", operation_Id)
    | join kind=inner requestMap on Ref
    | extend BucketStart = bin_at(timestamp, bucketSize, datetime(1970-01-01))
    | project BucketStart, itemCount, Entity = RequestEntity, Stage = RequestStage;
let requestsByBucket =
    reqs
    | summarize Requests = sum(itemCount) by BucketStart;
let exceptionsByBucket =
    stageExceptions
    | summarize Exceptions = sum(itemCount) by BucketStart;
let domainRows =
    requestsByBucket
    | join kind=fullouter exceptionsByBucket on BucketStart
    | extend BucketStart = coalesce(BucketStart, BucketStart1),
             Requests = coalesce(Requests, 0),
             Exceptions = coalesce(Exceptions, 0)
    | project ['kind'] = "bucket", BucketStart, TotalCount = tolong(Exceptions),
              Value = iff(Requests > 0, 100.0 * todouble(Exceptions) / todouble(Requests), real(null)),
              Timestamp = datetime(null), OperationId = "";
let entityRequests =
    reqs
    | summarize Requests = sum(itemCount), FailedCount = sumif(itemCount, success == false) by BucketStart, Entity;
let entityExceptions =
    stageExceptions
    | summarize Exceptions = sum(itemCount) by BucketStart, Entity;
let entityRows =
    entityRequests
    | join kind=fullouter entityExceptions on BucketStart, Entity
    | extend BucketStart = coalesce(BucketStart, BucketStart1),
             Entity = coalesce(Entity, Entity1),
             Requests = coalesce(Requests, 0),
             FailedCount = coalesce(FailedCount, 0),
             Exceptions = coalesce(Exceptions, 0)
    | sort by BucketStart asc, Requests desc, Exceptions desc, Entity asc
    | extend EntityRank = row_number(1, prev(BucketStart) != BucketStart)
    | where EntityRank <= maxEntities
    | project ['kind'] = "bucket", BucketStart, EntityKey = Entity,
              TotalCount = tolong(Exceptions),
              Value = iff(Requests > 0, 100.0 * todouble(Exceptions) / todouble(Requests), real(null)),
              P50 = real(null), P95 = real(null), Mean = real(null),
              FailedCount = tolong(FailedCount), Timestamp = datetime(null), OperationId = "";
{{grain}}Rows
