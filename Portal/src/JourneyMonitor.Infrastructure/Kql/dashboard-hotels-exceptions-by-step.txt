// Copied from Azure dashboard 6b35d70f tile "Hotel T-Mobile — exceptions by step".
let win = 4h;
let tmoHost = "stays.tmobiletravel.com";
exceptions
| where timestamp > ago(win)
| where type !startswith "ErrorEvent"
| join kind=inner (
    requests
    | where timestamp > ago(win)
    | extend Host = tostring(parse_url(url).Host)
    | where Host == tmoHost
    | extend StageNo = case(
        url has "search.json", 2,
        url has "BackgroundHotelSearch", 3,
        url has "searchrooms" or url has "getcancelpolicy" or url has "HotelDetails"
            or url has "ZentrumHubHotelDetailsContent" or url has "GetReviews", 4,
        url has "book.json", 5,
        url has "/reservations/Purchase" and url !has "PurchaseComplete"
            and url !has "CompletePurchase", 6,
        url has "userpeople_read" or url has "AddRemoveInsurance"
            or url has "TravelerProfile", 6,
        url has "CompletePurchase" or url has "determine3dseligibility", 7,
        url has "PurchaseComplete", 8,
        0)
    | where StageNo > 0
    | extend Stage = case(
        StageNo == 2, "02 Hotel Search",
        StageNo == 3, "03 Hotel Results",
        StageNo == 4, "04 Hotel Details",
        StageNo == 5, "05 PreBook",
        StageNo == 6, "06 Checkout",
        StageNo == 7, "07 Booking",
        "08 Confirmation")
    | distinct operation_Id, Stage
) on operation_Id
| summarize ExceptionCount = count() by bin(timestamp, 5m), Stage
