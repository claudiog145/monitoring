// Scout Monkey priority board - the invariant half.
//
// Nothing in this file names a company, a product area, a host, a journey step or a customer's
// word for any of them. Everything a company decides arrives through a placeholder that
// ScoutBoardQueryComposer fills from its CompanyProfile and the board request. Onboarding an
// enterprise is writing a profile; it is not editing this file.
//
// {{SOURCE:alias}} is the one placeholder left standing on purpose - IKqlSourceResolver turns each
// alias into an Application Insights ARM resource id later, against the registered catalogue.
//
// Shape kept from the workbook-derived board this replaces: bucket by time, summarise requests and
// exceptions per group, compare each bucket against the one before it, rank, then take the top
// rows PER ENTITY rather than globally - a global top-N lets the highest-volume entity crowd every
// other one off the board, which is how 112 affected entities went unseen on real data.
let serviceFilter={{SERVICE_FILTER}};
let entityFilter={{ENTITY_FILTER}};
let bucketSize={{BUCKET_MINUTES}}*1m;
let currentStart={{TIME_START}};
let tEnd={{TIME_END}};
let historyStart=bin(currentStart,bucketSize)-bucketSize;
// A row the profile does not describe is labelled, never dropped. Dropping it would shrink the
// board, and a board filtered down to nothing is indistinguishable from a healthy platform.
let unmappedService="(service not in profile)";
let unmappedStep="(step not in profile)";
let unknownEntity="(entity unavailable)";
let rawRequests=materialize(union
{{SOURCE_UNION}}
| extend Host=tolower(tostring(parse_url(url).Host)),Path=tolower(tostring(parse_url(url).Path))
| extend CorrelationId=coalesce(tostring(customDimensions.CorrelationId),tostring(customDimensions.correlationId),tostring(customDimensions.correlation_id))
| extend Entity=tostring({{ENTITY_IDENTITY}})
| extend Entity=case(isnotempty(Entity),Entity,isnotempty(Host),Host,unknownEntity)
| where {{HOST_FILTER}}
{{SERVICE_AND_STEP_CLASSIFIER}}
| extend Service=iff(isempty(Service),unmappedService,Service),Step=iff(isempty(Step),unmappedStep,Step));
let allRequests=materialize(rawRequests
| extend Ms=todouble(duration),IsFailure=success==false,Component=cloud_RoleName,Instance=cloud_RoleInstance,Ref=strcat(Source,":",operation_Id)
| extend CorrelationId=iff(CorrelationId=="00000000-0000-0000-0000-000000000000","",CorrelationId)
| extend UserKey=iff(AuthenticatedId in ("0","00000000-0000-0000-0000-000000000000"),"",AuthenticatedId),SessionKey=SessionId,BrowserKey=user_Id
| extend CorrKey=iff(isnotempty(CorrelationId),strcat("corr:",CorrelationId),strcat("op:",Ref)),GroupKey=strcat(Source," | ",ServiceKey," | ",Entity," | ",StepKey)
| where isempty(serviceFilter) or ServiceKey==serviceFilter
| where isempty(entityFilter) or Entity==entityFilter
| project timestamp,Source,ServiceKey,Service,StepKey,Step,Entity,Host,Ms,IsFailure,Component,Instance,CorrelationId,CorrKey,GroupKey,Ref,operation_Id,name,Path,resultCode,UserKey,BrowserKey,SessionKey);
let exceptionRows=union
{{EXCEPTION_SOURCE_UNION}};
let requestMap=allRequests
| where isnotempty(operation_Id)
| summarize arg_max(timestamp,GroupKey,ServiceKey,Service,StepKey,Step,Entity,CorrKey,CorrelationId,UserKey,BrowserKey,SessionKey) by Ref
| project Ref,RequestGroup=GroupKey,RequestServiceKey=ServiceKey,RequestService=Service,RequestStepKey=StepKey,RequestStep=Step,RequestEntity=Entity,RequestCorrKey=CorrKey,RequestCorrId=CorrelationId,RequestUser=UserKey,RequestBrowser=BrowserKey,RequestSession=SessionKey;
// An exception inherits the classification of the request it happened inside. Classifying it a
// second time, on its own, is how the board and the case file start telling different stories.
let mappedExceptions=materialize(exceptionRows
| extend Ref=strcat(Source,":",operation_Id),DirectCorr=coalesce(tostring(customDimensions.CorrelationId),tostring(customDimensions.correlationId),tostring(customDimensions.correlation_id))
| lookup kind=leftouter requestMap on Ref
| extend ServiceKey=coalesce(RequestServiceKey,""),Service=coalesce(RequestService,unmappedService),StepKey=coalesce(RequestStepKey,""),Step=coalesce(RequestStep,unmappedStep),Entity=coalesce(RequestEntity,unknownEntity)
| extend CorrelationId=coalesce(RequestCorrId,iff(DirectCorr=="00000000-0000-0000-0000-000000000000","",DirectCorr))
| extend CorrKey=coalesce(RequestCorrKey,iff(isnotempty(CorrelationId),strcat("corr:",CorrelationId),strcat("op:",Ref))),GroupKey=coalesce(RequestGroup,strcat(Source," | ",ServiceKey," | ",Entity," | ",StepKey))
| where isempty(serviceFilter) or ServiceKey==serviceFilter
| where isempty(entityFilter) or Entity==entityFilter
| extend UserKey=coalesce(RequestUser,""),SessionKey=coalesce(RequestSession,""),BrowserKey=coalesce(user_Id,RequestBrowser)
| project timestamp,Source,ServiceKey,Service,StepKey,Step,Entity,GroupKey,Ref,operation_Id,CorrKey,CorrelationId,type,outerMessage,innermostMessage,UserKey,BrowserKey,SessionKey);
let requestBins=allRequests
| summarize Requests=count(),MeanMs=avg(Ms),P95Ms=percentile(Ms,95),MaxMs=max(Ms),FailedRequests=countif(IsFailure),RequestUsers=make_set_if(UserKey,isnotempty(UserKey),40),RequestSessions=make_set_if(strcat(UserKey,"|",SessionKey),isnotempty(UserKey) and isnotempty(SessionKey),40),RequestCorrs=make_set_if(CorrelationId,isnotempty(CorrelationId),40),RequestOps=make_set_if(operation_Id,isnotempty(operation_Id),10),SlowOps=make_set_if(operation_Id,Ms>=10000 and isnotempty(operation_Id),10),SlowUsers=dcountif(UserKey,Ms>=10000 and isnotempty(UserKey)),SlowUserRecords=countif(Ms>=10000 and isnotempty(UserKey)),SlowRecords=countif(Ms>=10000),SlowBrowsers=dcountif(BrowserKey,Ms>=10000 and isempty(UserKey) and isnotempty(BrowserKey)),RequestPaths=make_set_if(Path,isnotempty(Path),10),RequestNames=make_set_if(name,isnotempty(name),10),SlowPaths=make_set_if(Path,Ms>=10000 and isnotempty(Path),10) by GroupKey,Bucket=bin(timestamp,bucketSize),Source,ServiceKey,Service,Entity,StepKey,Step;
let exceptionBins=mappedExceptions
| summarize Exceptions=count(),ExceptionUsersSet=make_set_if(UserKey,isnotempty(UserKey),40),ExceptionSessions=make_set_if(strcat(UserKey,"|",SessionKey),isnotempty(UserKey) and isnotempty(SessionKey),40),ExceptionCorrs=make_set_if(CorrelationId,isnotempty(CorrelationId),40),ExceptionOps=make_set_if(operation_Id,isnotempty(operation_Id),20),ExceptionUsers=dcountif(UserKey,isnotempty(UserKey)),ExceptionUserRecords=countif(isnotempty(UserKey)),ExceptionBrowsers=dcountif(BrowserKey,isempty(UserKey) and isnotempty(BrowserKey)),ExceptionTypes=make_set(type,3),ErrorExample=take_any(substring(coalesce(innermostMessage,outerMessage),0,180)) by GroupKey,Bucket=bin(timestamp,bucketSize),Source,ServiceKey,Service,Entity,StepKey,Step;
let metrics=materialize(requestBins | join kind=fullouter exceptionBins on GroupKey,Bucket,Source,ServiceKey,Service,Entity,StepKey,Step
| project GroupKey=coalesce(GroupKey,GroupKey1),Bucket=coalesce(Bucket,Bucket1),Source=coalesce(Source,Source1),ServiceKey=coalesce(ServiceKey,ServiceKey1),Service=coalesce(Service,Service1),Entity=coalesce(Entity,Entity1),StepKey=coalesce(StepKey,StepKey1),Step=coalesce(Step,Step1),Requests=coalesce(Requests,0),MeanMs,P95Ms,MaxMs,FailedRequests=coalesce(FailedRequests,0),Exceptions=coalesce(Exceptions,0),ExceptionTypes,ErrorExample,RequestOps,RequestPaths,RequestNames,SlowPaths,SlowOps,ExceptionOps,SlowUsers,SlowUserRecords,SlowRecords,SlowBrowsers,ExceptionUsers,ExceptionUserRecords,ExceptionBrowsers,RequestUsers,RequestSessions,RequestCorrs,ExceptionUsersSet,ExceptionSessions,ExceptionCorrs);
// The preceding bucket of the SAME group, not a global baseline: "slower than everyone else" and
// "slower than it was five minutes ago" are different claims and only the second one is actionable.
let compared=metrics
| lookup kind=leftouter (metrics | project GroupKey,Bucket=Bucket+bucketSize,PreviousP95=P95Ms,PreviousExceptions=Exceptions) on GroupKey,Bucket
| where Bucket>=currentStart and Bucket<tEnd;
let tops=materialize(compared
{{PRIORITY_MODEL}}
| extend Priority=case(PriorityRank==3,"Investigate now",PriorityRank==2,"Investigate next",PriorityRank==1,"Review","Low sample"),FailurePercent=round(FailedRate,1),LatencyGrowthX=round(LatencyRatio,2),ExceptionGrowthX=round(ExceptionRatio,2)
| partition hint.strategy=native by Entity (sort by {{PRIORITY_SORT}} | take {{ROWS_PER_ENTITY}}));
tops
| extend Start=Bucket,End=min_of(Bucket+bucketSize,tEnd),CaseId=strcat(GroupKey," | ",Bucket),IntervalStatus=iff(Bucket+bucketSize>tEnd,"Partial","Complete")
| extend OpSelection=tostring(set_union(coalesce(SlowOps,dynamic([])),coalesce(ExceptionOps,dynamic([])),coalesce(RequestOps,dynamic([])))),StartEpoch=tostring(datetime_diff("millisecond",Start,datetime(1970-01-01))),EndEpoch=tostring(datetime_diff("millisecond",End,datetime(1970-01-01)))
| extend GroupPayload=tostring(bag_pack("CaseId",CaseId,"GroupKey",GroupKey,"ServiceKey",ServiceKey,"Service",Service,"StepKey",StepKey,"Step",Step,"Entity",Entity,"Source",Source,"Start",Start,"End",End,"Priority",Priority,"Reason",Reason,"Ops",parse_json(OpSelection),"Users",set_union(coalesce(RequestUsers,dynamic([])),coalesce(ExceptionUsersSet,dynamic([]))),"Sessions",set_union(coalesce(RequestSessions,dynamic([])),coalesce(ExceptionSessions,dynamic([]))),"Correlations",set_union(coalesce(RequestCorrs,dynamic([])),coalesce(ExceptionCorrs,dynamic([]))),"Paths",coalesce(RequestPaths,dynamic([])),"Operations",coalesce(RequestNames,dynamic([])),"SlowPaths",coalesce(SlowPaths,dynamic([])),"ExceptionTypes",coalesce(ExceptionTypes,dynamic([])),"ErrorExample",ErrorExample))
| extend Change=case(isnull(PreviousP95),"No preceding interval",P95Ms>PreviousP95,"Increase",P95Ms<PreviousP95,"Decrease","Unchanged")
| project Priority,Reason,Interval=Start,Service,ServiceKey,Entity,Step,StepKey,P95Seconds=round(P95Ms/1000.0,3),PreviousP95Seconds=round(PreviousP95/1000.0,3),Change,MeanSeconds=round(MeanMs/1000.0,3),MaxSeconds=round(MaxMs/1000.0,3),Requests,FailedRequests,FailurePercent,LatencyGrowthX,Exceptions,ExceptionGrowthX,UniqueUserIds=iff(SlowUserRecords>0,SlowUsers,long(null)),AnonymousBrowserIds=SlowBrowsers,UserIdCoveragePct=iff(SlowRecords>0,round(100.0*SlowUserRecords/SlowRecords,1),real(null)),IntervalStatus,Source,GroupPayload,PriorityRank,OpSelection,StartEpoch,EndEpoch
| order by PriorityRank desc, P95Seconds desc, Exceptions desc
