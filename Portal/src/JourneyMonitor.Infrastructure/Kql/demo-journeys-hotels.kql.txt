// DEMO TELEMETRY ONLY - run against the demo stack's fake Application Insights (Insights:QueryEndpoint).
// TMobile_HOTEL_Query_KQL_v6d_demo_2h.txt matches the production tenant: T-Mobile hosts, catalog club
// ids, customDimensions.UserId/CorrelationId and "CompletePurchase result" traces. The tourism demo
// pack emits none of them (club9001.travel-demo.example hosts, VacationClubId 9001.., the user_Id and
// session_Id columns), so the legacy query returns 0 rows there. This query reads the demo shape and
// returns the legacy row contract, plus ClubId, so the hotel.* tables and the incumbent's Overview
// keep working in the demo.
// One journey row per session_Id; CorrelationId is session_Id-1 like the legacy rawId-seq key.
// Hotel steps 02-05 come from the hotel flow. The shared steps (01 Login, 06 Checkout, 07 Booking,
// 08 Confirmation, 09 Cancellation) are sessions of their own in the demo and are attributed to
// Hotels; car sessions (/rental/) are left to the Cars query. BookOk counts successful
// completepurchase calls in the session. Exceptions is FailedCalls (long).
// ClubId is the VacationClubId the demo telemetry carries (9001, 9002, ...), the same ids the
// Club filter lists, so the stored rows follow that filter.
// win and rowCap are replaced by the sync before the query runs.
let win = 15m;
let rowCap = 50;
let calls = materialize(
    requests
    | where timestamp > ago(win)
    | where isnotempty(session_Id)
    | extend Path = tolower(tostring(parse_url(url).Path))
    | where Path !contains "/rental/"
    | extend StageNo = case(
        Path contains "login.json" or Path contains "logincheck", 1,
        Path contains "backgroundhotelsearch", 3,
        Path contains "search.json", 2,
        Path contains "searchrooms" or Path contains "getcancelpolicy" or Path contains "hoteldetails"
            or Path contains "zentrumhubhoteldetailscontent" or Path contains "getreviews", 4,
        Path contains "book.json", 5,
        Path contains "purchasecomplete", 8,
        Path contains "completepurchase" or Path contains "determine3dseligibility", 7,
        Path contains "/reservations/purchase" or Path contains "userpeople_read"
            or Path contains "addremoveinsurance" or Path contains "travelerprofile", 6,
        Path contains "/reservations/cancel", 9,
        0)
    | where StageNo > 0
    | extend Stage = case(
        StageNo == 1, "01 Login",
        StageNo == 2, "02 Hotel Search",
        StageNo == 3, "03 Hotel Results",
        StageNo == 4, "04 Hotel Details",
        StageNo == 5, "05 PreBook",
        StageNo == 6, "06 Checkout",
        StageNo == 7, "07 Booking",
        StageNo == 8, "08 Confirmation",
        "09 Cancellation")
    | extend ClubId = tostring(customDimensions.VacationClubId)
    | project CorrelationId = strcat(session_Id, "-1"), UserId = user_Id,
              ClubId, StageNo, Stage, Kind = "api", Service = Path,
              Ms = todouble(duration), Failed = (success == false), ResultCode = tostring(resultCode),
              Ts = timestamp, operation_Id);
let vendorCalls =
    dependencies
    | where timestamp > ago(win)
    | join kind=inner (calls | project operation_Id, CorrelationId, UserId, ClubId, StageNo, Stage) on operation_Id
    | project CorrelationId, UserId, ClubId, StageNo, Stage, Kind = "vendor", Service = target,
              Ms = todouble(duration), Failed = (success == false), ResultCode = tostring(resultCode),
              Ts = timestamp, operation_Id;
let excByOp =
    exceptions
    | where timestamp > ago(win)
    | extend ExcMsg = substring(replace_regex(coalesce(outerMessage, ""), @"\s+", " "), 0, 140)
    | summarize ExcText = take_any(strcat(type, ": ", ExcMsg)) by operation_Id;
let allCalls = materialize(
    union calls, vendorCalls
    | join kind=leftouter excByOp on operation_Id
    | extend Detail = case(
        Failed == false, "",
        Kind == "api" and isnotempty(ExcText), strcat("HTTP ", ResultCode, " - ", ExcText),
        strcat("HTTP ", ResultCode))
    | project-away operation_Id1, ExcText);
let sessionFacts =
    allCalls
    | where Kind == "api"
    | summarize UserId = take_any(UserId),
                ClubId = take_any(ClubId),
                BookOk = countif(StageNo == 7 and Failed == false),
                BookFail = countif(StageNo == 7 and Failed),
                FailReason = take_anyif(Detail, StageNo == 7 and Failed and isnotempty(Detail))
      by CorrelationId;
let callMetrics =
    allCalls
    | summarize N = count(),
                CallTotalMs = round(sum(Ms)),
                P50Ms = round(percentile(Ms, 50)),
                P90Ms = round(percentile(Ms, 90)),
                P95Ms = round(percentile(Ms, 95)),
                P99Ms = round(percentile(Ms, 99)),
                MaxMs = round(max(Ms)),
                Fails = countif(Failed),
                Errors = make_list_if(bag_pack("n", 1, "t", Ts, "message", Detail), Failed and isnotempty(Detail), 8),
                LastTs = max(Ts),
                Ts = min(Ts)
      by CorrelationId, StageNo, Stage, Kind, Service
    | extend FailureRatePct = round(todouble(Fails) / N * 100.0, 2),
             Status = case(
                P95Ms < 2000, "GREEN",
                P95Ms < 5000, "YELLOW",
                "RED");
let stageApiMetrics =
    allCalls
    | where Kind == "api" and isnotnull(Ms)
    | summarize StageTotalMs = round(sum(Ms)),
                StepP50 = round(percentile(Ms, 50)),
                StepP90 = round(percentile(Ms, 90)),
                StepP95 = round(percentile(Ms, 95)),
                StepP99 = round(percentile(Ms, 99)),
                StepMax = round(max(Ms))
      by CorrelationId, StageNo, Stage;
let stageRows =
    callMetrics
    | sort by CorrelationId asc, StageNo asc, Kind asc, P95Ms desc, Service asc
    | summarize Calls = make_list(bag_pack(
            "kind", Kind, "call", Service, "t", Ts, "n", N,
            "callTotalMs", CallTotalMs, "p50Ms", P50Ms, "p90Ms", P90Ms,
            "p95Ms", P95Ms, "p99Ms", P99Ms, "maxMs", MaxMs,
            "failureRatePct", FailureRatePct, "status", Status,
            "detail", "", "errors", Errors), 80),
        StepStart = min(Ts),
        StepEnd = max(LastTs),
        CallCount = sum(N),
        StepFails = sum(Fails),
        StepTotal = sum(N)
      by CorrelationId, StageNo, Stage
    | join kind=leftouter stageApiMetrics on CorrelationId, StageNo, Stage
    | extend StepFailPct = iff(StepTotal > 0, round(todouble(StepFails) / StepTotal * 100.0, 2), real(null)),
             StepStatus = case(
                isnull(StepP95), "N/A",
                StepP95 < 2000, "GREEN",
                StepP95 < 5000, "YELLOW",
                "RED");
stageRows
| sort by CorrelationId asc, StageNo asc
| summarize Journey = make_list(bag_pack(
        "stageNo", StageNo, "stage", Stage, "t", StepStart,
        "callCount", CallCount, "stageTotalMs", StageTotalMs,
        "p50Ms", StepP50, "p90Ms", StepP90, "p95Ms", StepP95,
        "p99Ms", StepP99, "maxMs", StepMax,
        "failureRatePct", StepFailPct, "status", StepStatus, "calls", Calls), 14),
    Steps = make_list(Stage, 14),
    LastStep = max(StageNo),
    FailedCalls = sum(StepFails),
    JourneyTotalMs = round(sum(todouble(coalesce(StageTotalMs, 0.0)))),
    SlowestCallMs = max(StepMax),
    OverallP95 = max(StepP95),
    SearchAt = minif(StepStart, StageNo == 2),
    PurchaseAt = minif(StepStart, StageNo == 7),
    ConfirmAt = minif(StepStart, StageNo == 8),
    CancelAt = minif(StepStart, StageNo == 9),
    FirstSeen = min(StepStart),
    LastSeen = max(StepEnd)
  by CorrelationId
| join kind=inner sessionFacts on CorrelationId
| project-away CorrelationId1
| where isnotempty(UserId)
| extend DistinctOkRes = iff(BookOk > 0, 1, 0),
         BookedReservations = dynamic([]),
         SessionCount = 1,
         CallsViaCorrelation = 0,
         CallsNoCorrelation = 0,
         IsBookSuccessful = BookOk > 0,
         PassedSearch = isnotnull(SearchAt)
| extend RetryNote = iff(BookOk > 0 and BookFail > 0,
                         strcat(" after ", tostring(BookFail), " failed attempt", iff(BookFail > 1, "s", "")),
                         "")
| extend Outcome = case(
    BookOk > 0 and isnotnull(CancelAt), strcat("Booked", RetryNote, "; cancellation activity"),
    BookOk > 0 and isnotnull(ConfirmAt), strcat("Booked", RetryNote),
    BookOk > 0, strcat("Booked", RetryNote, ", no confirmation page"),
    BookFail > 0, strcat("Purchase failed x", tostring(BookFail)),
    isnotnull(CancelAt), "Cancellation activity",
    isnotnull(ConfirmAt), "Confirmation page",
    isnotnull(PurchaseAt), "Purchase attempted, no result",
    strcat("Dropped at step ", tostring(LastStep)))
| extend SearchToConfirm_s = datetime_diff("second", ConfirmAt, SearchAt)
| project UserId, Outcome, LastStep, Steps, FailReason,
          CorrelationId, SessionCount, CallsViaCorrelation, CallsNoCorrelation,
          BookOk, BookFail, DistinctOkRes, BookedReservations, FailedCalls, IsBookSuccessful, PassedSearch, Exceptions = FailedCalls, JourneyTotalMs,
          SlowestCallMs, OverallP95, SearchToConfirm_s,
          FirstSeen, LastSeen, Journey, ClubId
| sort by ClubId asc, LastSeen desc
| extend _Rank = row_number(1, prev(ClubId) != ClubId)
| where _Rank <= rowCap
| project-away _Rank
